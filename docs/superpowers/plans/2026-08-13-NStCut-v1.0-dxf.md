- [ ] **Step 2: DXF geometry reader**

Create `src/NStCut.Dxf/Services/DxfGeometryService.cs`:

```csharp
using netDxf;
using netDxf.Entities;
using NStCut.Core.Contracts;
using NStCut.Core.Geometry;

namespace NStCut.Dxf.Services;

public sealed class DxfGeometryService : IDxfGeometryService
{
    public DxfContour ReadContour(string path)
    {
        if (!File.Exists(path))
            throw new FileNotFoundException("DXF not found.", path);

        var doc = DxfDocument.Load(path);
        var points = new List<Point>();
        var bounds = new Rect(0, 0, 0, 0);
        bool hasBounds = false;

        foreach (var entity in doc.Entities.GetAllEntities())
        {
            switch (entity)
            {
                case LwPolyline lw:
                    foreach (var v in lw.Vertices)
                    {
                        AddPoint(ref bounds, ref hasBounds, v.Location.X, v.Location.Y);
                    }
                    break;

                case Line line:
                    AddPoint(ref bounds, ref hasBounds, line.StartPoint.X, line.StartPoint.Y);
                    AddPoint(ref bounds, ref hasBounds, line.EndPoint.X, line.EndPoint.Y);
                    break;

                case Arc arc:
                    var arcBounds = ArcBounds(arc);
                    AddRect(ref bounds, ref hasBounds, arcBounds);
                    break;

                case Circle circle:
                    AddPoint(ref bounds, ref hasBounds, circle.Center.X - circle.Radius, circle.Center.Y - circle.Radius);
                    AddPoint(ref bounds, ref hasBounds, circle.Center.X + circle.Radius, circle.Center.Y + circle.Radius);
                    break;
            }
        }

        return new DxfContour
        {
            Bounds = hasBounds ? bounds : new Rect(0, 0, 0, 0),
            Polygon = null // polygonal contour will be added in v1.1
        };
    }

    private static Rect ArcBounds(Arc arc)
    {
        // Simplified: use start, end and center extremes
        var samples = 12;
        var minX = double.MaxValue;
        var maxX = double.MinValue;
        var minY = double.MaxValue;
        var maxY = double.MinValue;

        for (int i = 0; i <= samples; i++)
        {
            double angle = arc.StartAngle + (arc.EndAngle - arc.StartAngle) * i / samples;
            double rad = Math.PI * angle / 180.0;
            double x = arc.Center.X + arc.Radius * Math.Cos(rad);
            double y = arc.Center.Y + arc.Radius * Math.Sin(rad);

            minX = Math.Min(minX, x);
            maxX = Math.Max(maxX, x);
            minY = Math.Min(minY, y);
            maxY = Math.Max(maxY, y);
        }

        return new Rect(minX, minY, maxX - minX, maxY - minY);
    }

    private static void AddPoint(ref Rect bounds, ref bool hasBounds, double x, double y)
    {
        if (!hasBounds)
        {
            bounds = new Rect(x, y, 0, 0);
            hasBounds = true;
            return;
        }

        var minX = Math.Min(bounds.Left, x);
        var maxX = Math.Max(bounds.Right, x);
        var minY = Math.Min(bounds.Bottom, y);
        var maxY = Math.Max(bounds.Top, y);
        bounds = new Rect(minX, minY, maxX - minX, maxY - minY);
    }

    private static void AddRect(ref Rect bounds, ref bool hasBounds, Rect rect)
    {
        AddPoint(ref bounds, ref hasBounds, rect.Left, rect.Bottom);
        AddPoint(ref bounds, ref hasBounds, rect.Right, rect.Top);
    }
}
```

- [ ] **Step 3: DXF export writer**

Create `src/NStCut.Dxf/Services/DxfExportService.cs`:

```csharp
using netDxf;
using netDxf.Entities;
using netDxf.Tables;
using NStCut.Core.Contracts;
using NStCut.Core.Geometry;
using NStCut.Core.Models;

namespace NStCut.Dxf.Services;

public sealed class DxfExportService : IDxfExportService
{
    public void WriteSheet(string outputPath, SheetInstance sheet, double partGapMm, double sheetMarginMm)
    {
        var doc = new DxfDocument();

        // Sheet boundary rectangle
        var boundary = new LwPolyline(new List<LwPolylineVertex>
        {
            new(new netDxf.Vector2(0, 0)),
            new(new netDxf.Vector2(sheet.Material.Length, 0)),
            new(new netDxf.Vector2(sheet.Material.Length, sheet.Material.Width)),
            new(new netDxf.Vector2(0, sheet.Material.Width))
        })
        {
            IsClosed = true
        };
        boundary.Layer = new Layer("SheetBoundary") { Color = AciColor.Yellow };
        doc.Entities.Add(boundary);

        foreach (var placed in sheet.Parts)
        {
            WritePlacedPart(doc, placed);
        }

        Directory.CreateDirectory(Path.GetDirectoryName(outputPath)!);
        doc.Save(outputPath);
    }

    private static void WritePlacedPart(DxfDocument targetDoc, PlacedPart placed)
    {
        if (!File.Exists(placed.Source.DxfPath))
            return;

        var sourceDoc = DxfDocument.Load(placed.Source.DxfPath);
        var sourceBounds = placed.Source.Bounds;

        foreach (var entity in sourceDoc.Entities.GetAllEntities())
        {
            var clone = (EntityObject)entity.Clone();

            // Translate so the source bounding-box bottom-left lands at (0,0),
            // then rotate, then move to final position.
            // netDxf uses column vectors: M_final * v. Apply transforms right-to-left.
            var toOrigin = netDxf.Matrix3.Translation(new netDxf.Vector3(-sourceBounds.Left, -sourceBounds.Bottom, 0));
            var rotation = netDxf.Matrix3.RotationZ(placed.Rotation * Math.PI / 180.0);
            var toPlace = netDxf.Matrix3.Translation(new netDxf.Vector3(placed.Position.X, placed.Position.Y, 0));

            clone.TransformBy(toPlace * rotation * toOrigin);
            clone.Layer = new Layer("Parts") { Color = AciColor.White };
            targetDoc.Entities.Add(clone);
        }
    }
}
```

- [ ] **Step 4: Unit tests**

Create `src/NStCut.Dxf.Tests/DxfServicesTests.cs`:

```csharp
using NStCut.Core.Geometry;
using NStCut.Core.Models;
using NStCut.Dxf.Services;

namespace NStCut.Dxf.Tests;

public class DxfServicesTests : IDisposable
{
    private readonly string _tempDir = Path.Combine(Path.GetTempPath(), Guid.NewGuid().ToString("N"));

    public DxfServicesTests()
    {
        Directory.CreateDirectory(_tempDir);
    }

    public void Dispose()
    {
        if (Directory.Exists(_tempDir))
            Directory.Delete(_tempDir, true);
    }

    [Fact]
    public void ReadContour_ReturnsBoundsFromRectanglePolyline()
    {
        var path = Path.Combine(_tempDir, "rect.dxf");
        CreateRectangleDxf(path, 10, 20);

        var service = new DxfGeometryService();
        var contour = service.ReadContour(path);

        Assert.Equal(10, contour.Bounds.Width);
        Assert.Equal(20, contour.Bounds.Height);
    }

    [Fact]
    public void WriteSheet_CreatesOutputFile()
    {
        var sourcePath = Path.Combine(_tempDir, "part.dxf");
        CreateRectangleDxf(sourcePath, 10, 10);

        var material = new SheetMaterial { Name = "Steel", Length = 100, Width = 50 };
        var sheet = new SheetInstance { Material = material, Index = 1 };
        sheet.Parts.Add(new PlacedPart
        {
            Source = new Part { Name = "P1", DxfPath = sourcePath, Bounds = new Rect(0, 0, 10, 10) },
            CopyIndex = 1,
            Position = new Point(5, 5),
            Rotation = 0,
            Bounds = new Rect(5, 5, 10, 10)
        });

        var export = new DxfExportService();
        var outPath = Path.Combine(_tempDir, "sheet.dxf");
        export.WriteSheet(outPath, sheet, 5, 5);

        Assert.True(File.Exists(outPath));
        Assert.True(new FileInfo(outPath).Length > 0);
    }

    private static void CreateRectangleDxf(string path, double width, double height)
    {
        var doc = new netDxf.DxfDocument();
        var rect = new netDxf.Entities.LwPolyline(new List<netDxf.Entities.LwPolylineVertex>
        {
            new(new netDxf.Vector2(0, 0)),
            new(new netDxf.Vector2(width, 0)),
            new(new netDxf.Vector2(width, height)),
            new(new netDxf.Vector2(0, height))
        })
        {
            IsClosed = true
        };
        doc.Entities.Add(rect);
        doc.Save(path);
    }
}
```
