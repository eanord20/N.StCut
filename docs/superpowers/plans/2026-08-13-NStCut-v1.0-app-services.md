```csharp
using System.Net.Http;
using System.Text.Json;
using System.Reflection;

namespace NStCut.App.Services;

public sealed class UpdateService
{
    private readonly HttpClient _httpClient = new();

    public async Task<bool> IsUpdateAvailableAsync(Version currentVersion)
    {
        try
        {
            var response = await _httpClient.GetStringAsync(
                "https://api.github.com/repos/eanord20/windows-tool/releases/latest");
            using var doc = JsonDocument.Parse(response);
            var tag = doc.RootElement.GetProperty("tag_name").GetString() ?? string.Empty;
            if (tag.StartsWith("v", StringComparison.OrdinalIgnoreCase))
                tag = tag[1..];
            if (Version.TryParse(tag, out var latest))
                return latest > currentVersion;
        }
        catch
        {
            // ignore network/parsing errors for MVP skeleton
        }
        return false;
    }

    public Version CurrentVersion =>
        Assembly.GetEntryAssembly()?.GetName().Version ?? new Version(1, 0, 0, 0);
}
```

```csharp
using System.Text;
using NStCut.Core.Models;

namespace NStCut.App.Services;

public sealed class FileResolveService
{
    public IReadOnlyList<string> GetCandidates(string partName, string searchPath)
    {
        if (!Directory.Exists(searchPath) || string.IsNullOrWhiteSpace(partName))
            return Array.Empty<string>();

        var expectedFull = NameNormalizer.Normalize(partName);
        var expectedPrefix = NameNormalizer.FirstToken(partName);

        var allFiles = Directory
            .EnumerateFiles(searchPath, "*.dxf", SearchOption.AllDirectories)
            .ToList();

        // 1) exact normalized match
        var candidates = allFiles
            .Where(path => NameNormalizer.Normalize(Path.GetFileNameWithoutExtension(path)) == expectedFull)
            .ToList();

        if (candidates.Count > 0)
            return candidates;

        // 2) fallback: first token of part name is prefix of file name
        if (!string.IsNullOrWhiteSpace(expectedPrefix))
        {
            candidates = allFiles
                .Where(path => NameNormalizer.Normalize(Path.GetFileNameWithoutExtension(path))
                    .StartsWith(expectedPrefix, StringComparison.OrdinalIgnoreCase))
                .ToList();
        }

        return candidates;
    }

    public ResolveResult Resolve(string partName, string searchPath)
    {
        var candidates = GetCandidates(partName, searchPath);
        return candidates.Count switch
        {
            0 => new ResolveResult(PartStatus.NotFound, string.Empty, candidates),
            1 => new ResolveResult(PartStatus.Found, candidates[0], candidates),
            _ => new ResolveResult(PartStatus.Ambiguous, string.Empty, candidates)
        };
    }
}

public sealed record ResolveResult(PartStatus Status, string Path, IReadOnlyList<string> Candidates);

public static class NameNormalizer
{
    public static string Normalize(string input)
    {
        if (string.IsNullOrWhiteSpace(input)) return string.Empty;

        var sb = new StringBuilder();
        bool prevWasSpace = false;

        foreach (var ch in input.Trim().ToLowerInvariant())
        {
            if (IsForbiddenWindowsChar(ch))
                continue;

            if (ch == '_' || ch == '-' || char.IsWhiteSpace(ch))
            {
                if (!prevWasSpace)
                {
                    sb.Append(' ');
                    prevWasSpace = true;
                }
                continue;
            }

            sb.Append(ch);
            prevWasSpace = false;
        }

        return sb.ToString().Trim();
    }

    public static string FirstToken(string input)
    {
        var normalized = Normalize(input);
        return normalized.Split(' ', StringSplitOptions.RemoveEmptyEntries).FirstOrDefault() ?? string.Empty;
    }

    private static bool IsForbiddenWindowsChar(char ch)
        => ch is '<' or '>' or ':' or '"' or '/' or '\\' or '|' or '?' or '*';
}
```

```csharp
using System.Text.Json;
using NStCut.App.Models;

namespace NStCut.App.Services;

public sealed class ProjectFileService
{
    private static readonly JsonSerializerOptions SerializerOptions = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase
    };

    public void Save(string path, ProjectData data)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        File.WriteAllText(path, JsonSerializer.Serialize(data, SerializerOptions));
    }

    public ProjectData Load(string path)
    {
        var json = File.ReadAllText(path);
        return JsonSerializer.Deserialize<ProjectData>(json, SerializerOptions) ?? new ProjectData();
    }
}
```

```csharp
using System.ComponentModel;
using System.Runtime.CompilerServices;
using NStCut.Core.Models;

namespace NStCut.App.Models;

public sealed class PartRow : INotifyPropertyChanged
{
    private int _quantity = 1;
    private string _name = string.Empty;
    private string _materialId = string.Empty;
    private string _dxfPath = string.Empty;
    private PartStatus _status = PartStatus.Pending;
    private bool _isManuallyResolved;
    private GrainMode _grainMode = GrainMode.FollowMaterial;
    private int _rowNumber;

    public event PropertyChangedEventHandler? PropertyChanged;

    public int RowNumber
    {
        get => _rowNumber;
        set { _rowNumber = value; OnPropertyChanged(); }
    }

    public int Quantity
    {
        get => _quantity;
        set { _quantity = Math.Max(1, value); OnPropertyChanged(); }
    }

    public string Name
    {
        get => _name;
        set { _name = value ?? string.Empty; OnPropertyChanged(); }
    }

    public string MaterialId
    {
        get => _materialId;
        set { _materialId = value ?? string.Empty; OnPropertyChanged(); }
    }

    public string DxfPath
    {
        get => _dxfPath;
        set { _dxfPath = value ?? string.Empty; OnPropertyChanged(); }
    }

    public PartStatus Status
    {
        get => _status;
        set { _status = value; OnPropertyChanged(); }
    }

    public bool IsManuallyResolved
    {
        get => _isManuallyResolved;
        set { _isManuallyResolved = value; OnPropertyChanged(); }
    }

    public GrainMode GrainMode
    {
        get => _grainMode;
        set { _grainMode = value; OnPropertyChanged(); OnPropertyChanged(nameof(CanRotate)); }
    }

    public bool CanRotate
    {
        get => GrainMode == GrainMode.Allow90;
        set
        {
            GrainMode = value ? GrainMode.Allow90 : GrainMode.FollowMaterial;
            OnPropertyChanged();
            OnPropertyChanged(nameof(GrainMode));
        }
    }

    public Part ToPart()
    {
        return new Part
        {
            Name = Name,
            Quantity = Quantity,
            MaterialId = MaterialId,
            DxfPath = DxfPath,
            Status = Status,
            IsManuallyResolved = IsManuallyResolved,
            GrainMode = GrainMode
        };
    }

    public static PartRow FromPart(Part part)
    {
        return new PartRow
        {
            Name = part.Name,
            Quantity = part.Quantity,
            MaterialId = part.MaterialId,
            DxfPath = part.DxfPath,
            Status = part.Status,
            IsManuallyResolved = part.IsManuallyResolved,
            GrainMode = part.GrainMode
        };
    }

    public PartRow Clone()
    {
        return new PartRow
        {
            RowNumber = RowNumber,
            Quantity = Quantity,
            Name = Name,
            MaterialId = MaterialId,
            DxfPath = DxfPath,
            Status = Status,
            IsManuallyResolved = IsManuallyResolved,
            GrainMode = GrainMode
        };
    }

    private void OnPropertyChanged([CallerMemberName] string? propertyName = null)
        => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
}
```
