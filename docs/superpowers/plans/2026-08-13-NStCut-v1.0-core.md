```csharp
namespace NStCut.Core.Geometry;

public readonly record struct Rect(double X, double Y, double Width, double Height)
{
    public double Left => X;
    public double Right => X + Width;
    public double Bottom => Y;
    public double Top => Y + Height;
    public double Area => Width * Height;

    public bool Intersects(Rect other)
    {
        return Left < other.Right && Right > other.Left &&
               Bottom < other.Top && Top > other.Bottom;
    }

    public bool Contains(Rect other)
    {
        return Left <= other.Left && Right >= other.Right &&
               Bottom <= other.Bottom && Top >= other.Top;
    }
}
```

Create `src/NStCut.Core/Geometry/Polygon2D.cs`:

```csharp
using System.Linq;

namespace NStCut.Core.Geometry;

public sealed class Polygon2D
{
    public IReadOnlyList<Point> Vertices { get; }

    public Polygon2D(IEnumerable<Point> vertices)
    {
        Vertices = vertices.ToList().AsReadOnly();
    }

    public Rect Bounds
    {
        get
        {
            if (Vertices.Count == 0) return new Rect(0, 0, 0, 0);
            var minX = Vertices.Min(v => v.X);
            var maxX = Vertices.Max(v => v.X);
            var minY = Vertices.Min(v => v.Y);
            var maxY = Vertices.Max(v => v.Y);
            return new Rect(minX, minY, maxX - minX, maxY - minY);
        }
    }
}
```

```csharp
using NStCut.Core.Geometry;

namespace NStCut.Core.Models;

public sealed class Part
{
    public string Name { get; set; } = string.Empty;
    public int Quantity { get; set; } = 1;
    public string MaterialId { get; set; } = string.Empty;
    public string DxfPath { get; set; } = string.Empty;
    public PartStatus Status { get; set; } = PartStatus.Pending;
    public bool IsManuallyResolved { get; set; }
    public GrainMode GrainMode { get; set; } = GrainMode.FollowMaterial;
    public Polygon2D? Contour { get; set; }
    public Rect Bounds { get; set; }
    public double Area { get; set; }

    public Part Clone()
    {
        return new Part
        {
            Name = Name,
            Quantity = Quantity,
            MaterialId = MaterialId,
            DxfPath = DxfPath,
            Status = Status,
            IsManuallyResolved = IsManuallyResolved,
            GrainMode = GrainMode,
            Contour = Contour is null ? null : new Polygon2D(Contour.Vertices),
            Bounds = Bounds,
            Area = Area
        };
    }
}
```

```csharp
using NStCut.Core.Geometry;

namespace NStCut.Core.Tests;

public class GeometryTests
{
    [Fact]
    public void Rect_Area_IsCorrect()
    {
        var rect = new Rect(0, 0, 10, 20);
        Assert.Equal(200, rect.Area);
    }

    [Fact]
    public void Rect_Intersects_ReturnsTrueForOverlapping()
    {
        var a = new Rect(0, 0, 10, 10);
        var b = new Rect(5, 5, 10, 10);
        Assert.True(a.Intersects(b));
    }

    [Fact]
    public void Rect_Contains_ReturnsTrueWhenFullyInside()
    {
        var outer = new Rect(0, 0, 100, 100);
        var inner = new Rect(10, 10, 80, 80);
        Assert.True(outer.Contains(inner));
    }

    [Fact]
    public void Polygon_Bounds_ComputesMinMax()
    {
        var poly = new Polygon2D([new Point(1, 2), new Point(5, 8), new Point(3, 0)]);
        var bounds = poly.Bounds;
        Assert.Equal(1, bounds.Left);
        Assert.Equal(5, bounds.Right);
        Assert.Equal(0, bounds.Bottom);
        Assert.Equal(8, bounds.Top);
    }
}
```
