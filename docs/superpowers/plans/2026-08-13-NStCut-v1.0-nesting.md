- [ ] **Step 1: Orientation helper**

Create `src/NStCut.Nesting/Internals/OrientationHelper.cs`:

```csharp
using NStCut.Core.Geometry;
using NStCut.Core.Models;

namespace NStCut.Nesting.Internals;

internal static class OrientationHelper
{
    public static IEnumerable<(double Width, double Height)> GetOrientations(Part part, SheetMaterial material)
    {
        var w = part.Bounds.Width;
        var h = part.Bounds.Height;

        var allowRotation = part.GrainMode == GrainMode.Allow90 ||
                            (part.GrainMode == GrainMode.FollowMaterial && !material.HasGrain);

        yield return (w, h);

        if (allowRotation && Math.Abs(w - h) > 0.001)
            yield return (h, w);
    }
}
```

- [ ] **Step 2: Shelf packer**

Create `src/NStCut.Nesting/Internals/ShelfPacker.cs`:

```csharp
using NStCut.Core.Geometry;
using NStCut.Core.Models;

namespace NStCut.Nesting.Internals;

internal sealed class ShelfPacker
{
    private readonly SheetMaterial _material;
    private readonly double _margin;
    private readonly double _gap;
    private readonly double _usableWidth;
    private readonly double _usableHeight;

    private double _shelfY;
    private double _shelfHeight;
    private double _cursorX;

    public List<PlacedPart> Parts { get; } = new();
    public double PlacedArea { get; private set; }

    public ShelfPacker(SheetMaterial material, double margin, double gap)
    {
        _material = material;
        _margin = margin;
        _gap = gap;
        _usableWidth = material.Length - 2 * margin;
        _usableHeight = material.Width - 2 * margin;
        _shelfY = 0;
        _shelfHeight = 0;
        _cursorX = 0;
    }

    public bool TryPlace(Part part, int copyIndex)
    {
        foreach (var (w, h) in OrientationHelper.GetOrientations(part, _material))
        {
            var effectiveW = w + _gap;
            var effectiveH = h + _gap;

            if (effectiveW > _usableWidth || effectiveH > _usableHeight)
                continue;

            if (_cursorX + effectiveW > _usableWidth)
            {
                if (_shelfY + _shelfHeight + effectiveH > _usableHeight)
                    continue;

                _shelfY += _shelfHeight;
                _shelfHeight = 0;
                _cursorX = 0;
            }

            var placed = new PlacedPart
            {
                Source = part,
                CopyIndex = copyIndex,
                Position = new Point(_margin + _cursorX, _margin + _shelfY),
                Rotation = Math.Abs(w - part.Bounds.Width) < 0.001 ? 0 : 90,
                Bounds = new Rect(_margin + _cursorX, _margin + _shelfY, w, h)
            };

            Parts.Add(placed);
            PlacedArea += w * h;

            _cursorX += effectiveW;
            if (effectiveH > _shelfHeight)
                _shelfHeight = effectiveH;

            return true;
        }

        return false;
    }

    public SheetInstance ToSheet(int index)
    {
        var sheet = new SheetInstance
        {
            Material = _material,
            Index = index,
        };
        sheet.Parts.AddRange(Parts);

        var sheetArea = _material.Length * _material.Width;
        sheet.WasteArea = sheetArea - PlacedArea;
        sheet.UtilizationPercent = sheetArea > 0 ? PlacedArea / sheetArea * 100.0 : 0.0;
        return sheet;
    }
}
```

- [ ] **Step 3: Engine implementation**

Create `src/NStCut.Nesting/Services/BoundingBoxShelfEngine.cs`:

```csharp
using NStCut.Core.Contracts;
using NStCut.Core.Models;
using NStCut.Nesting.Internals;

namespace NStCut.Nesting.Services;

public sealed class BoundingBoxShelfEngine : INestingEngine
{
    public NestingResult Nest(
        IReadOnlyList<Part> parts,
        IReadOnlyList<SheetMaterial> materials,
        NestingOptions options,
        IProgress<double>? progress = null,
        CancellationToken cancellationToken = default)
    {
        var result = new NestingResult();
        var materialDict = materials.ToDictionary(m => m.Id);

        var expanded = parts
            .SelectMany(p => Enumerable.Range(1, Math.Max(1, p.Quantity))
                .Select(i => (CopyIndex: i, Part: p)))
            .ToList();

        var groups = expanded.GroupBy(x => x.Part.MaterialId);
        int processed = 0;
        double totalParts = expanded.Count;

        foreach (var group in groups)
        {
            cancellationToken.ThrowIfCancellationRequested();

            if (!materialDict.TryGetValue(group.Key, out var material))
            {
                foreach (var item in group)
                    result.Unplaced.Add(CloneForUnplaced(item.Part));
                continue;
            }

            var copies = group.ToList();
            copies.Sort((a, b) => b.Part.Bounds.Area.CompareTo(a.Part.Bounds.Area));

            var packers = new List<ShelfPacker> { new(material, options.SheetMarginMm, options.PartGapMm) };

            foreach (var (copyIndex, part) in copies)
            {
                bool placed = false;
                foreach (var packer in packers)
                {
                    if (packer.TryPlace(part, copyIndex))
                    {
                        placed = true;
                        break;
                    }
                }

                if (!placed)
                {
                    var newPacker = new ShelfPacker(material, options.SheetMarginMm, options.PartGapMm);
                    if (newPacker.TryPlace(part, copyIndex))
                    {
                        packers.Add(newPacker);
                    }
                    else
                    {
                        result.Unplaced.Add(CloneForUnplaced(part));
                    }
                }

                processed++;
                if (totalParts > 0)
                    progress?.Report(processed / totalParts * 100.0);
            }

            int sheetIndex = 1;
            foreach (var packer in packers)
            {
                if (packer.Parts.Count > 0)
                    result.Sheets.Add(packer.ToSheet(sheetIndex++));
            }
        }

        result.TotalUtilizationPercent = result.Sheets.Any()
            ? result.Sheets.Average(s => s.UtilizationPercent)
            : 0.0;

        return result;
    }

    private static Part CloneForUnplaced(Part part)
    {
        var clone = part.Clone();
        clone.Quantity = 1;
        return clone;
    }
}
```

- [ ] **Step 4: Unit tests**

Create `src/NStCut.Nesting.Tests/BoundingBoxShelfEngineTests.cs`:

```csharp
using NStCut.Core.Constants;
using NStCut.Core.Geometry;
using NStCut.Core.Models;
using NStCut.Nesting.Services;

namespace NStCut.Nesting.Tests;

public class BoundingBoxShelfEngineTests
{
    private static Part CreatePart(string name, double w, double h, string materialId, int qty = 1)
    {
        return new Part
        {
            Name = name,
            Quantity = qty,
            MaterialId = materialId,
            Bounds = new Rect(0, 0, w, h),
            Area = w * h,
            GrainMode = GrainMode.Allow90
        };
    }

    [Fact]
    public void Nest_FitsAllOnOneSheet()
    {
        var engine = new BoundingBoxShelfEngine();
        var material = new SheetMaterial { Name = "Steel", Length = 100, Width = 100 };
        var parts = new List<Part>
        {
            CreatePart("A", 40, 40, material.Id),
            CreatePart("B", 40, 40, material.Id)
        };

        var result = engine.Nest(parts, new List<SheetMaterial> { material }, DefaultNestingOptions.Value);

        Assert.Single(result.Sheets);
        Assert.Equal(2, result.Sheets[0].Parts.Count);
        Assert.Empty(result.Unplaced);
    }

    [Fact]
    public void Nest_CreatesSecondSheetWhenNeeded()
    {
        var engine = new BoundingBoxShelfEngine();
        var material = new SheetMaterial { Name = "Steel", Length = 50, Width = 100 };
        var parts = new List<Part>
        {
            CreatePart("A", 40, 80, material.Id),
            CreatePart("B", 40, 80, material.Id)
        };

        var result = engine.Nest(parts, new List<SheetMaterial> { material }, DefaultNestingOptions.Value);

        Assert.Equal(2, result.Sheets.Count);
        Assert.Empty(result.Unplaced);
    }

    [Fact]
    public void Nest_MarksOversizedPartAsUnplaced()
    {
        var engine = new BoundingBoxShelfEngine();
        var material = new SheetMaterial { Name = "Steel", Length = 10, Width = 10 };
        var parts = new List<Part>
        {
            CreatePart("A", 20, 20, material.Id)
        };

        var result = engine.Nest(parts, new List<SheetMaterial> { material }, DefaultNestingOptions.Value);

        Assert.Empty(result.Sheets);
        Assert.Single(result.Unplaced);
    }

    [Fact]
    public void Nest_RotationReducesSheetCount()
    {
        var engine = new BoundingBoxShelfEngine();
        var material = new SheetMaterial { Name = "Steel", Length = 60, Width = 40 };
        var parts = new List<Part>
        {
            CreatePart("A", 50, 20, material.Id)
        };

        var noRotation = parts.Select(p => { p.GrainMode = GrainMode.Fixed0; return p; }).ToList();
        var withRotation = parts.Select(p => { p.GrainMode = GrainMode.Allow90; return p; }).ToList();

        var r1 = engine.Nest(noRotation, new List<SheetMaterial> { material }, DefaultNestingOptions.Value);
        var r2 = engine.Nest(withRotation, new List<SheetMaterial> { material }, DefaultNestingOptions.Value);

        Assert.Equal(2, r1.Sheets.Count);
        Assert.Single(r2.Sheets);
    }
}
```
